async function fetchWeather() { try { // Fetch location details via IP const locRes = await fetch('https://ipapi.co/json/'); const locData = await locRes.json(); const city = locData.city || locData.region || "London"; document.getElementById('city').innerText = city.toUpperCase(); // Fetch live weather data from wttr.in (uses local station observations) const weatherRes = await fetch(`https://wttr.in/${encodeURIComponent(city)}?format=j1`); const weatherData = await weatherRes.json(); const currentCondition = weatherData.current_condition[0]; const todayForecast = weatherData.weather[0]; const tempC = currentCondition.temp_C; const conditionText = currentCondition.weatherDesc[0].value; const maxTemp = todayForecast.maxtempC; const minTemp = todayForecast.mintempC; // Map condition text to appropriate emoji let icon = "⛅"; const descLower = conditionText.toLowerCase(); if (descLower.includes("sun") || descLower.includes("clear")) icon = "☀️"; else if (descLower.includes("cloud") || descLower.includes("overcast")) icon = "☁️"; else if (descLower.includes("rain") || descLower.includes("drizzle")) icon = "🌧️"; else if (descLower.includes("snow") || descLower.includes("sleet")) icon = "❄️"; else if (descLower.includes("thunder")) icon = "⛈️"; else if (descLower.includes("fog") || descLower.includes("mist")) icon = "🌫️"; // Update HTML elements document.getElementById('temp').innerText = `${tempC}°`; document.getElementById('condition').innerText = conditionText; document.getElementById('icon').innerText = icon; document.getElementById('range').innerText = `H: ${maxTemp}° L: ${minTemp}°`; } catch (err) { document.getElementById('city').innerText = "LONDON"; document.getElementById('temp').innerText = "18°"; document.getElementById('condition').innerText = "Partly Cloudy"; document.getElementById('icon').innerText = "⛅"; document.getElementById('range').innerText = "H: 21° L: 14°"; } } fetchWeather();
